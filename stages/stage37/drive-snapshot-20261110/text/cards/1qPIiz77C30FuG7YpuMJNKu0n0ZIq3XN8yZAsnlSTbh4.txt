function sieve(N){
 let prime=Array(N+1).fill(true);prime[0]=prime[1]=false;
 for(let p=2;p*p<=N;p++)if(prime[p])for(let m=p*p;m<=N;m+=p)prime[m]=false;
 return prime;
}
function isKfull(N,k){
 let n=Math.abs(N);if(n===1)return true;
 for(let p=2;p*p<=n;p++)if(n%p===0){
  let e=0;while(n%p===0){e++;n/=p;}
  if(e<k)return false;
 }
 return n===1; // a remaining prime factor has exponent 1 (<k)
}
