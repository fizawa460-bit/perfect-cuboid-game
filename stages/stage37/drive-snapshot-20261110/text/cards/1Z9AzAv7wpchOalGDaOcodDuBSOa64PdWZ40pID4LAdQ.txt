L=[d,a*d+b*c,b*d-a*c,(b*b-a*a)*c+2*a*b*d]
P2=a*a+b*b
identities={
 'rotation_norm':L[1]**2+L[2]**2-P2*(c*c+d*d),
 'inverse_d':a*L[1]+b*L[2]-P2*d,
 'inverse_c':b*L[1]-a*L[2]-P2*c,
 'fourth_form':b*L[1]+a*L[2]-L[3],
 'p_mod_fourth':L[3]-2*b*L[1]+P2*c,
}
for name,expr in identities.items():
    assert s.expand(expr)==0,name
coeff=[(s.diff(v,c),s.diff(v,d)) for v in L]
dets=[s.expand(u[0]*v[1]-u[1]*v[0])
      for i,u in enumerate(coeff) for v in coeff[i+1:]]
assert all(s.expand(u-v)==0 for u,v in zip(dets,[-b,a,a*a-b*b,P2,a*P2,-b*P2]))
norm_coeff=[s.expand(u*u+v*v) for u,v in coeff]
assert all(s.expand(u-v)==0 for u,v in zip(norm_coeff,[1,P2,P2,P2**2]))

quad2=b*ell[1]+a*ell[2]-ell[3]
