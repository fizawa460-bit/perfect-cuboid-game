from math import gcd,isqrt
from collections import defaultdict,Counter
def census(B):
  legs=defaultdict(set)
  for m in range(2,isqrt(B)+2):
    for n in range(1,m):
      if gcd(m,n)!=1 or (m-n)%2==0: continue
      h=m*m+n*n
      if h>B: continue
      a,b=m*m-n*n,2*m*n
      for k in range(1,B//h+1):
        legs[k*a].add(k*b);legs[k*b].add(k*a)
  points=[]
  for e,S in legs.items():
    L=sorted(S)
    for i,x in enumerate(L):
      for y in L[i:]:
        z=e*e+x*x+y*y;R=isqrt(z)
        if R<=B and R*R==z and gcd(gcd(e,x),y)==1:
          U=isqrt(e*e+x*x); V=isqrt(e*e+y*y)
          W=U*V-e*R
          assert W>0 and U*U==e*e+x*x and V*V==e*e+y*y
          points.append((e,x,y,U,V,R,W,gcd(e*R,x*y)))
  return points
