\[

L^\times/(L^\times)^2\mathbf Q^\times
\]

\[
\chi_{p,k}(a-b\theta)=
\left(\frac{\sum_{j=0}^{d_k} f_{k,j} a^j b^{d_k-j}}{p}\right).
\]
