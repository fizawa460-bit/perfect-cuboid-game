def gf_rank(vectors):
    piv={}
    for z in vectors:
        while z:
            h=z.bit_length()-1
            if h in piv:z ^= piv[h]
            else:piv[h]=z;break
    return len(piv)

# Enumerate cubic integer rings; x is a uniformizer at the unique prime over 2.
# Cubic KA has x^3 - 8 x^2 + 6x + 2, cubic KB y^3+4y^2-6y+2.
class Cubic2:
    def __init__(self,poly):
        self.c=tuple(poly)
        self.qsq=None
        self.coset_table=None
        self.labels=None
    def mul(self,a,b):
        v=[0]*5
        for i in range(3):
            for j in range(3):v[i+j]+=a[i]*b[j]
        for i in (4,3):
            q=v[i]
            if q:
                for j in range(3):v[i-3+j]-=q*self.c[j]
        return tuple(v[:3])
    def divide_by_uniformizer(self,a):
        # pi^3+c2*pi^2+c1*pi+2=0; 2/pi=-(pi^2+c2*pi+c1)
        twice_inv=(-self.c[1],-self.c[2],-1)
        z=self.mul(a,twice_inv)
        assert all(zi%2==0 for zi in z)
        return tuple(zi//2 for zi in z)
    def norm(self,a):
        cols=[]
        for j in range(3):
            e=[0,0,0];e[j]=1
            cols.append(self.mul(a,e))
        M=sp.Matrix(3,3,lambda i,j:cols[j][i])
        return int(M.det())
    def square_residues(self):
        if self.qsq is None:
            self.qsq={tuple(v%8 for v in self.mul(c,c)) for c in product(range(8),repeat=3) if c[0]%2}
            assert len(self.qsq)==16
        return self.qsq
    def normalized(self,a):
        n=self.norm(a)
        assert n!=0
        e=v_p(abs(n),2)
        for _ in range(e):a=self.divide_by_uniformizer(a)
        assert a[0]%2
        return e%2,tuple(t%8 for t in a)
    def setup_labels(self):
        squares=self.square_residues(); coset_table={}; reps=[]
        for u in product(range(8),repeat=3):
            if not u[0]%2 or u in coset_table:continue
            cid=len(reps)
            orbit={tuple(t%8 for t in self.mul(u,s)) for s in squares}
            for t in orbit:coset_table[t]=cid
            reps.append(min(orbit))
        assert len(reps)==16 and len(coset_table)==256
        # obtain a group-basis via orbit expansion, not by copying older labels
        labs={coset_table[(1,0,0)]:0};gen=[]
        for cid in range(16):
            if cid in labs:continue
            val=reps[cid]; k=len(gen)
            snapshot=list(labs.items()); gen.append(val)
            for old,mask in snapshot:
                pair=self.mul(reps[old],val)
                nxt=coset_table[tuple(v%8 for v in pair)]
                assert nxt not in labs
                labs[nxt]=mask^(1<<k)
        assert len(gen)==4 and len(labs)==16
        self.coset_table=coset_table; self.labels=labs
    def word(self,a):
        if self.labels is None:self.setup_labels()
        parity,residue=self.normalized(a)
        return parity<<4 | self.labels[self.coset_table[residue]]

# Pure modular 5-adic finite field arithmetic over monic irreducible mod-5 polys.
class F5Ext:
    def __init__(self, coeffs, mod):
        self.poly=tuple(v%5 for v in mod);self.d=len(mod)-1
        self.c=tuple((list(coeffs)+[0]*self.d)[:self.d]);self.c=tuple(t%5 for t in self.c)
    def coerce(self,r):
        if isinstance(r,F5Ext):
            assert r.poly==self.poly
            return r
        return F5Ext([r],self.poly)
    def __add__(self,r):
        r=self.coerce(r);return F5Ext([(u+v)%5 for u,v in zip(self.c,r.c)],self.poly)
    __radd__=__add__
    def __neg__(self):return F5Ext([-v for v in self.c],self.poly)
    def __sub__(self,r):return self+-self.coerce(r)
    def __rsub__(self,r):return self.coerce(r)-self
    def __mul__(self,r):
        r=self.coerce(r);c=[0]*(2*self.d-1)
        for i,a in enumerate(self.c):
            for j,b in enumerate(r.c):c[i+j]+=a*b
        for k in range(len(c)-1,self.d-1,-1):
            for j in range(self.d):c[k-self.d+j]-=c[k]*self.poly[j]
        return F5Ext(c[:self.d],self.poly)
    __rmul__=__mul__
    def __pow__(self,n):
        assert n>=0
        cur=self;out=self.coerce(1)
        while n:
            if n%2:out=out*cur
            cur=cur*cur;n//=2
        return out
    def inv(self):return self**(5**self.d-2)
    def __truediv__(self,r):return self*self.coerce(r).inv()
    def __rtruediv__(self,r):return self.coerce(r)/self
    def __eq__(self,r):return self.c==self.coerce(r).c
    def char(self):
        assert self!=0
        val=self**((5**self.d-1)//2)
        assert val==1 or val==-1
        return int(val==-1)
